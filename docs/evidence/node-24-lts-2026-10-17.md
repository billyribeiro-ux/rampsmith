The check script:

```js
import { execFileSync } from 'node:child_process';
import semver from 'semver';

const [nodeVersion, typesVersion, ...specs] = process.argv.slice(2);
let failures = 0;
for (const spec of specs) {
  // `name` and `version` always exist, so npm prints an object even when engines and peers are absent.
  const raw = execFileSync('npm', ['view', spec, 'name', 'version', 'engines', 'peerDependencies', '--json'], {
    encoding: 'utf8',
  });
  const info = JSON.parse(raw);
  if (Array.isArray(info) || typeof info !== 'object' || info === null || typeof info.version !== 'string') {
    throw new Error(`${spec} did not resolve to exactly one version`);
  }
  const engineRange = info.engines?.node;
  const typesRange = info.peerDependencies?.['@types/node'];
  const engineOk = engineRange === undefined || semver.satisfies(nodeVersion, engineRange);
  const typesOk = typesRange === undefined || semver.satisfies(typesVersion, typesRange);
  if (!engineOk || !typesOk) failures += 1;
  const columns = [
    engineOk && typesOk ? 'OK  ' : 'FAIL',
    `${spec} (registry ${info.version})`,
    `engines.node=${engineRange ?? 'not declared'}`,
  ];
  if (typesRange !== undefined) columns.push(`peer @types/node=${typesRange}`);
  console.log(columns.join('\t'));
}
console.log(failures === 0 ? `ALL ${specs.length} OK for Node ${nodeVersion} and @types/node ${typesVersion}` : `${failures} of ${specs.length} FAILED`);
process.exit(failures === 0 ? 0 : 1);
```
